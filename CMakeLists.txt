cmake_minimum_required(VERSION 3.20)
project(AmberArcade VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(arcade_engine src/game.cpp src/render.cpp)
target_include_directories(arcade_engine PUBLIC src)
if(MSVC)
    target_compile_options(arcade_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(arcade_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_executable(AmberArcade WIN32 src/main.cpp src/platform_windows.cpp)
    target_compile_definitions(AmberArcade PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(AmberArcade PRIVATE arcade_engine gdi32 user32 winmm)
    if(MSVC)
        target_link_options(AmberArcade PRIVATE /ENTRY:mainCRTStartup)
    elseif(MINGW)
        target_link_options(AmberArcade PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
else()
    add_executable(AmberArcade src/main.cpp)
    target_link_libraries(AmberArcade PRIVATE arcade_engine)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(arcade_tests tests/tests.cpp)
    target_link_libraries(arcade_tests PRIVATE arcade_engine)
    add_test(NAME arcade_rules_and_waves COMMAND arcade_tests)
endif()
